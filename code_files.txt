# ============================================================
# Server (Логика)
# ============================================================
server <- function(input, output, session) {
  
  # Реактивный фильтр данных
  filtered_data <- reactive({
    d <- df
    
    if (input$district != "ALL") {
      d <- d %>% filter(district == input$district)
    }
    
    if (input$case_type != "ALL") {
      d <- d %>% filter(case_type == input$case_type)
    }
    
    # Фильтрация по выбранному паттерну
    if (input$pattern != "ALL") {
      # .data[[input$pattern]] позволяет динамически обращаться к колонке по её имени
      d <- d %>% filter(.data[[input$pattern]] == TRUE)
    }
    
    d <- d %>% filter(duration >= input$duration[1] & duration <= input$duration[2])
    
    d <- d %>% filter(n_violations >= input$n_violations[1] & n_violations <= input$n_violations[2])
    
    if (input$only_children) {
      d <- d %>% filter(kw_children == TRUE)
    }
    
    if (input$only_incident) {
      d <- d %>% filter(kw_incident == TRUE)
    }
    
    d
  })
  
  # Ключевые метрики
  output$total_cases <- renderText({
    nrow(filtered_data())
  })
  
  output$avg_duration <- renderText({
    sprintf("%.1f", mean(filtered_data()$duration, na.rm = TRUE))
  })
  
  output$avg_violations <- renderText({
    sprintf("%.1f", mean(filtered_data()$n_violations, na.rm = TRUE))
  })
  
  output$children_pct <- renderText({
    pct <- mean(filtered_data()$kw_children, na.rm = TRUE) * 100
    sprintf("%.1f%%", pct)
  })
  
  output$violations_pct <- renderText({
    pct <- mean(filtered_data()$n_violations > 0, na.rm = TRUE) * 100
    sprintf("%.1f%%", pct)
  })
  
  # Гистограмма длительности
  output$hist_duration <- renderPlot({
    ggplot(filtered_data(), aes(x = duration)) +
      geom_histogram(binwidth = 10, fill = "steelblue", color = "white", alpha = 0.85) +
      labs(title = "Распределение длительности рассмотрения дел",
           x = "Длительность (дни)", y = "Количество дел") +
      theme_minimal(base_size = 14)
  })
  
  # Столбчатая диаграмма типов дел
  output$bar_case_types <- renderPlot({
    type_data <- filtered_data() %>% count(case_type)
    ggplot(type_data, aes(x = reorder(case_type, -n), y = n)) +
      geom_col(fill = "coral", alpha = 0.85) +
      labs(title = "Распределение по типам", x = NULL, y = "Количество дел") +
      theme_minimal(base_size = 14) +
      theme(axis.text.x = element_text(angle = 30, hjust = 1))
  })
  
  # График паттернов
  output$bar_patterns <- renderPlot({
    pat_summary <- tibble(
      "Доказательства" = sum(filtered_data()$pat_evidence, na.rm = TRUE),
      "Хранение" = sum(filtered_data()$pat_storage, na.rm = TRUE),
      "Нет камер" = sum(filtered_data()$pat_absence_camera, na.rm = TRUE),
      "СанПиН" = sum(filtered_data()$pat_violation_sanpin, na.rm = TRUE),
      "Утечка" = sum(filtered_data()$pat_data_leak, na.rm = TRUE),
      "Неисправность" = sum(filtered_data()$pat_malfunction, na.rm = TRUE)
    ) %>% pivot_longer(everything(), names_to = "Паттерн", values_to = "Количество")
    
    ggplot(pat_summary, aes(x = reorder(Паттерн, -Количество), y = Количество)) +
      geom_col(fill = "darkorange", alpha = 0.85) +
      labs(title = "Паттерны нарушений", x = NULL, y = "Количество дел") +
      theme_minimal(base_size = 14) +
      theme(axis.text.x = element_text(angle = 30, hjust = 1))
  })
  
  # Boxplot по округам
  output$box_districts <- renderPlot({
    ggplot(filtered_data(), aes(x = reorder(district, duration, FUN = median),
                                y = duration, fill = district)) +
      geom_boxplot(alpha = 0.8) +
      coord_flip() +
      labs(title = "Длительность по округам", x = NULL, y = "Дни") +
      theme_minimal(base_size = 14) +
      guides(fill = "none")
  })
  
  # Ключевые темы
  output$keywords <- renderPlot({
    kw_summary <- tibble(
      "Видеонаблюдение" = sum(filtered_data()$kw_camera, na.rm = TRUE),
      "Школа" = sum(filtered_data()$kw_school, na.rm = TRUE),
      "Дети" = sum(filtered_data()$kw_children, na.rm = TRUE),
      "Инциденты" = sum(filtered_data()$kw_incident, na.rm = TRUE),
      "Отсутствие" = sum(filtered_data()$kw_absence, na.rm = TRUE),
      "Обязательства" = sum(filtered_data()$kw_obligate, na.rm = TRUE)
    ) %>% pivot_longer(everything(), names_to = "Тема", values_to = "Количество")
    
    ggplot(kw_summary, aes(x = reorder(Тема, -Количество), y = Количество)) +
      geom_col(fill = "seagreen3", alpha = 0.85) +
      labs(title = "Ключевые темы (слова)", x = NULL, y = "Количество дел") +
      theme_minimal(base_size = 14) +
      theme(axis.text.x = element_text(angle = 30, hjust = 1))
  })
  
  # Таблица данных
  output$data_table <- renderDT({
    datatable(
      filtered_data() %>%
        select(id, district, court, case_type, date_in, date_out, duration, n_violations, 
               pat_evidence, pat_storage, pat_absence_camera) %>%
        mutate(
          date_in = format(date_in, "%d.%m.%Y"),
          date_out = format(date_out, "%d.%m.%Y")
        ),
      options = list(pageLength = 10, scrollX = TRUE),
      rownames = FALSE
    ) %>%
      formatStyle('duration', backgroundColor = styleInterval(
        c(30, 60), c('white', 'lightyellow', 'lightblue')
      )) %>%
      formatStyle('n_violations', backgroundColor = styleInterval(
        c(0, 2), c('white', 'lightyellow', 'lightsalmon')
      ))
  })
}

# Запуск приложения
shinyApp(ui, server)

pattern_cols <- grep("^pat_", colnames(df), value = TRUE)

  warning("Колонки паттернов (pat_*) не найдены. Проверьте extract_features.R")
} else {
  cat(sprintf("Найдено %d паттернов: %s\n", length(pattern_cols), paste(pattern_cols, collapse=", ")))

# ============================================================
# Функция-генератор графиков (сохраняет в подпапку внутри eda_plots)
# ============================================================
plot_and_save <- function(data, base_dir, folder_name) {
  save_dir <- file.path(base_dir, folder_name)
  dir.create(save_dir, showWarnings = FALSE, recursive = TRUE)
  
  if (nrow(data) == 0) {
    cat(sprintf("  [Пропуск] %s: нет данных для графиков\n", folder_name))
    return(invisible())
  }
  
  cat(sprintf("  -> Генерация графиков для: %s (папка: %s)\n", folder_name, save_dir))
  
  custom_theme <- theme_minimal(base_size = 14) +
    theme(
      panel.grid.major = element_line(color = "grey70", linewidth = 0.5),
      panel.grid.minor = element_line(color = "grey85", linewidth = 0.25, linetype = "dashed"),
      panel.border = element_rect(color = "grey50", fill = NA, linewidth = 0.8)
    )
  
  # --- 1. Гистограмма длительности ---
  dur_data <- data %>% filter(!is.na(duration), duration < 365)
  if (nrow(dur_data) > 0) {
    max_dur <- max(dur_data$duration, na.rm = TRUE)
    step <- ifelse(max_dur <= 50, 5, ifelse(max_dur <= 150, 10, 20))
    
    p1 <- ggplot(dur_data, aes(x = duration)) +
      geom_histogram(binwidth = step, fill = "steelblue", color = "white", alpha = 0.85) +
      scale_x_continuous(breaks = seq(0, max_dur + step, by = step)) +
      labs(title = paste0(folder_name, ": Длительность рассмотрения дел"),
           x = "Дни", y = "Количество дел") + 
      custom_theme
    
    print(p1)
    ggsave(file.path(save_dir, "hist_duration.png"), p1, width = 8, height = 5, dpi = 150)
  }
  
  # --- 2. Столбчатая диаграмма типов дел ---
  type_data <- data %>% 
    count(case_type) %>%
    filter(!is.na(case_type), 
           !str_detect(case_type, "Не определено|ошибка загрузки|файл обрезан"))
  
  if (nrow(type_data) > 0) {
    max_n <- max(type_data$n, na.rm = TRUE)
    step_n <- ifelse(max_n <= 20, 2, ifelse(max_n <= 100, 10, 25))
    
    p2 <- ggplot(type_data, aes(x = reorder(case_type, -n), y = n)) +
      geom_col(fill = "coral", alpha = 0.85) +
      scale_y_continuous(breaks = seq(0, max_n + step_n, by = step_n)) +
      labs(title = paste0(folder_name, ": Типы судебных дел"), x = NULL, y = "Количество") +
      custom_theme + 
      theme(axis.text.x = element_text(angle = 30, hjust = 1))
    
    print(p2)
    ggsave(file.path(save_dir, "bar_case_types.png"), p2, width = 8, height = 5, dpi = 150)
  }
  
  # --- 3. Boxplot по округам (только для ALL) ---
  if (folder_name == "ALL" && length(unique(data$district)) > 1) {
    box_data <- data %>% filter(!is.na(duration), duration < 365)
    if (nrow(box_data) > 0) {
      max_dur <- max(box_data$duration, na.rm = TRUE)
      step <- ifelse(max_dur <= 50, 5, ifelse(max_dur <= 150, 10, 20))
      
      p3 <- ggplot(box_data, aes(x = reorder(district, duration, FUN = median), 
                                 y = duration, fill = district)) +
        geom_boxplot(alpha = 0.8) + 
        scale_y_continuous(breaks = seq(0, max_dur + step, by = step)) +
        coord_flip() + 
        guides(fill = "none") +
        labs(title = "Сравнение длительности по федеральным округам", 
             x = NULL, y = "Дни") + 
        custom_theme
      
      print(p3)
      ggsave(file.path(save_dir, "box_districts.png"), p3, width = 9, height = 6, dpi = 150)
    }
  }
  
  # --- 4. Распределение по ключевым словам ---
  kw_summary <- tibble(
    "Видеонаблюдение" = sum(data$kw_camera, na.rm = TRUE),
    "Школа/Образование" = sum(data$kw_school, na.rm = TRUE),
    "Дети/Несовершеннолетние" = sum(data$kw_children, na.rm = TRUE),
    "Инциденты" = sum(data$kw_incident, na.rm = TRUE),
    "Отсутствие оснащения" = sum(data$kw_absence, na.rm = TRUE),
    "Обязательные требования" = sum(data$kw_obligate, na.rm = TRUE)
  ) %>% pivot_longer(everything(), names_to = "Категория", values_to = "Количество")
  
  if (sum(kw_summary$Количество) > 0) {
    max_kw <- max(kw_summary$Количество, na.rm = TRUE)
    step_kw <- ifelse(max_kw <= 20, 2, ifelse(max_kw <= 100, 10, 25))
    
    p5 <- ggplot(kw_summary, aes(x = reorder(Категория, -Количество), y = Количество)) +
      geom_col(fill = "seagreen3", alpha = 0.85) +
      scale_y_continuous(breaks = seq(0, max_kw + step_kw, by = step_kw)) +
      labs(title = paste0(folder_name, ": Упоминания ключевых тем"), x = NULL, y = "Количество дел") +
      custom_theme +
      theme(axis.text.x = element_text(angle = 30, hjust = 1))
    
    print(p5)
    ggsave(file.path(save_dir, "keywords.png"), p5, width = 9, height = 5, dpi = 150)
  }
  
  # --- 5. Распределение паттернов правонарушений ---
  if (length(pattern_cols) > 0) {
    pat_data <- data %>%
      summarise(across(all_of(pattern_cols), ~ sum(.x, na.rm = TRUE))) %>%
      pivot_longer(everything(), names_to = "Паттерн", values_to = "Количество") %>%
      filter(Количество > 0)
    
    if (nrow(pat_data) > 0) {
      max_pat <- max(pat_data$Количество, na.rm = TRUE)
      step_pat <- ifelse(max_pat <= 20, 2, ifelse(max_pat <= 100, 10, 25))
      
      p6 <- ggplot(pat_data, aes(x = reorder(Паттерн, -Количество), y = Количество)) +
        geom_col(fill = "darkorange", alpha = 0.85) +
        scale_y_continuous(breaks = seq(0, max_pat + step_pat, by = step_pat)) +
        labs(title = paste0(folder_name, ": Паттерны правонарушений"), x = NULL, y = "Количество дел") +
        custom_theme +
        theme(axis.text.x = element_text(angle = 30, hjust = 1))
      
      print(p6)
      ggsave(file.path(save_dir, "patterns.png"), p6, width = 10, height = 5, dpi = 150)
    }
  }
  
  # --- 6. Тепловая карта: регион × паттерн (только для ALL) ---
  if (folder_name == "ALL" && length(pattern_cols) > 0 && length(unique(data$district)) > 1) {
    heat_data <- data %>%
      group_by(district) %>%
      summarise(across(all_of(pattern_cols), ~ mean(.x, na.rm = TRUE) * 100)) %>%
      pivot_longer(-district, names_to = "Паттерн", values_to = "Процент")
    
    p7 <- ggplot(heat_data, aes(x = Паттерн, y = district, fill = Процент)) +
      geom_tile(color = "white") +
      geom_text(aes(label = sprintf("%.0f%%", Процент)), size = 3) +
      scale_fill_gradient(low = "white", high = "red") +
      labs(title = "Доля дел с паттерном по федеральным округам, %", x = NULL, y = NULL) +
      theme_minimal(base_size = 12) +
      theme(axis.text.x = element_text(angle = 30, hjust = 1))
    
    print(p7)
    ggsave(file.path(save_dir, "heatmap_patterns.png"), p7, width = 10, height = 6, dpi = 150)
  }
}

# ============================================================
# Функция: доля дел с паттерном по федеральным округам (%)
# Возвращает таблицу, печатает копируемый TSV и сохраняет CSV
# ============================================================
pattern_share_by_district <- function(data,
                                      base_dir = BASE_DIR,
                                      pattern_cols = NULL,
                                      save_csv = TRUE,
                                      round_digits = 1) {
  # Если паттерны не переданы — берём все колонки, начинающиеся на pat_
  if (is.null(pattern_cols)) {
    pattern_cols <- grep("^pat_", colnames(data), value = TRUE)
  }
  if (length(pattern_cols) == 0) {
    warning("Колонки паттернов (pat_*) не найдены.")
    return(invisible(NULL))
  }
  
  # Считаем долю (в %) по каждому округу, плюс строку ALL (вся база)
  shares <- data %>%
    group_by(district) %>%
    summarise(
      n_cases = n(),
      across(all_of(pattern_cols), ~ round(mean(.x, na.rm = TRUE) * 100, round_digits)),
      .groups = "drop"
    )
  
  shares_all <- data %>%
    summarise(
      district = "ALL",
      n_cases = n(),
      across(all_of(pattern_cols), ~ round(mean(.x, na.rm = TRUE) * 100, round_digits))
    )
  
  result <- bind_rows(shares, shares_all)
  
  # --- Печать копируемой таблицы (TSV: разделитель — табуляция) ---
  cat("\n=== Доля дел с паттерном по федеральным округам, % ===\n")
  cat("(скопируйте и вставьте в Excel — разделитель табуляции)\n\n")
  
  # Заголовок
  cat(paste(colnames(result), collapse = "\t"), "\n", sep = "")
  # Строки
  for (i in seq_len(nrow(result))) {
    cat(paste(unlist(result[i, ]), collapse = "\t"), "\n", sep = "")
  }
  cat("\n")
  
  # --- Сохранение в CSV ---
  if (save_csv) {
    out_path <- file.path(base_dir, "data", "pattern_share_by_district.csv")
    dir.create(dirname(out_path), showWarnings = FALSE, recursive = TRUE)
    readr::write_csv(result, out_path)
    cat(sprintf("CSV сохранён: %s\n", out_path))
  }
  
  invisible(result)
}

pattern_share_by_district(df)

extract_one <- function(path) {
  district <- basename(dirname(path)) %>% str_remove("_Videonablyudenie_v_skole$")
  cid <- tools::file_path_sans_ext(basename(path))
  
  # 1. Читаем HTML
  html <- tryCatch(read_html(path, encoding = "UTF-8"), error = function(e) NULL)
  if (is.null(html)) return(NULL)
  
  # 2. БЕЗОПАСНОЕ удаление скриптов, стилей и рекламы перед подсчетом длины текста
  tryCatch({
    html %>% 
      html_elements("script, style, noscript, .ad-banner, .translate, #translator") %>% 
      xml2::xml_remove()
  }, error = function(e) NULL) # Игнорируем ошибку, если таких элементов нет
  
  # 3. Извлекаем чистый текст
  text <- html %>% html_text2()
  text_lower <- tolower(text)
  text_length <- nchar(text)
  
  # 4. ПРОВЕРКА НА "ОБОЛОЧКУ" (если текст слишком короткий, значит это пустая страница с рекламой)
  if (text_length < 3000) {
    meta_desc <- html %>% html_element("meta[name='description']") %>% html_attr("content")
    
    return(tibble(
      id = cid, 
      district = district, 
      is_valid = FALSE,
      court = ifelse(!is.na(meta_desc), str_extract(meta_desc, "^[^,]+"), "Не найден"),
      case_type = "Не определено (файл обрезан или пуст)",
      date_in = NA_character_, 
      date_out = str_extract(meta_desc, "\\d{2}\\.\\d{2}\\.\\d{4}"),
      text_length = text_length,
      kw_camera = FALSE, kw_school = FALSE, kw_children = FALSE,
      kw_incident = FALSE, kw_absence = FALSE, kw_obligate = FALSE,
      n_camera = 0, n_kamera = 0, n_school = 0, n_children = 0
    ))
  }
  
  # 5. Извлечение СУДА
  court_match <- str_match(text, "Суд:\\s*(.+)")
  court <- if (!is.na(court_match[1, 1])) str_trim(court_match[1, 2]) else "Не найден"
  
  # 6. Извлечение ДАТ
  date_in_match <- str_match(text, "Дата начала:\\s*(\\d{2}\\.\\d{2}\\.\\d{4})")
  date_in <- if (!is.na(date_in_match[1, 1])) date_in_match[1, 2] else NA_character_
  
  date_out_match <- str_match(text, "Дата рассмотрения:\\s*(\\d{2}\\.\\d{2}\\.\\d{4})")
  date_out <- if (!is.na(date_out_match[1, 1])) date_out_match[1, 2] else NA_character_
  
  # Если дата рассмотрения не найдена в блоке, ищем в тексте (например, "изготовлено 25 июня 2026 года")
  if (is.na(date_out)) {
    m_out <- str_match(text, "(?:изготовлен|вынесен|принят|опубликован).*?(\\d{1,2})\\s+(января|февраля|марта|апреля|мая|июня|июля|августа|сентября|октября|ноября|декабря)\\s+(\\d{4})")
    if (!is.na(m_out[1, 1])) {
      day <- m_out[1, 2]
      month_str <- m_out[1, 3]
      year <- m_out[1, 4]
      
      month_num <- switch(month_str,
                          "января" = "01", "февраля" = "02", "марта" = "03", "апреля" = "04",
                          "мая" = "05", "июня" = "06", "июля" = "07", "августа" = "08",
                          "сентября" = "09", "октября" = "10", "ноября" = "11", "декабря" = "12")
      date_out <- sprintf("%02d.%s.%s", as.integer(day), month_num, year)
    }
  }
  
  # 7. Тип дела
  case_type <- case_when(
    str_detect(text_lower, "административн.*(?:иск|дело|производство)") ~ "Административное",
    str_detect(text_lower, "уголовн.*(?:дело|производство)") ~ "Уголовное",
    str_detect(text_lower, "гражданск.*(?:дело|производство|иск)") ~ "Гражданское",
    str_detect(text_lower, "коап|об административных правонарушениях") ~ "КоАП",
    TRUE ~ "Не определено"
  )
  
  # 8. Ключевые слова
  kw_camera   <- str_detect(text_lower, "видеонаблюдени|камер")
  kw_school   <- str_detect(text_lower, "школ|образовательн")
  kw_children <- str_detect(text_lower, "дет|школьник|подрост|несовершеннолетн|учащ")
  kw_incident <- str_detect(text_lower, "инцидент|происшеств|драк|нападен|травм")
  kw_absence  <- str_detect(text_lower, "отсутств|не\\s+установлен|не\\s+оснащен|неисправн")
  kw_obligate <- str_detect(text_lower, "обязать|возложить обязанность")
  pat_malfunction <- str_detect(text_lower, "неисправн|сломан|не работ|поврежден|вышел из строя|не функционирует|не подключен|отключен|не эксплуатируется")
  pat_privacy <-     str_detect(text_lower, "раздевалк|туалет|санузл|приватн|личн(ая|ой|ую) жизн|жилое помещение|место отдыха")
  pat_storage <- str_detect(text_lower, "утер|хранени.*запис|несанкционированн|доступ к запис|архив|срок хранения|уничтожен")
  pat_evidence <- str_detect(text_lower, "доказательств|видеофиксаци|запис[ьи] с камер|подтвержд|видеозапис|материал.*видео")
  pat_no_act <- str_detect(text_lower, "отсутстви(е|я|и) положени|нет локальн.*акт|не принят приказ|регламент.*видеонаблюдени")
  pat_absence_camera <- str_detect(text_lower, "отсутств.*видеонаблюдени|не установлен.*камер|нет камер|не оснащен.*видеонаблюдени")
  pat_improper_placement <- str_detect(text_lower, "направлен.*на (окна|жилые|раздевалк|туалет)|расположен.*неправильно|нарушени.*расположени|видеонаблюдени.*в (раздевалк|туалет)")
  pat_no_consent <- str_detect(text_lower, "согласи.*на обработку|персональн.*данн.*дет|согласи.*родител|без согласия")
  pat_violation_sanpin <- str_detect(text_lower, "санпин|санитарн.*норм|роспотребнадзор|нарушени.*требован.*безопасности")
  pat_failure_to_install <- str_detect(text_lower, "не исполнен.*предписани|не выполн.*требовани.*установк|обязан.*установить.*камер")
  pat_data_leak <- str_detect(text_lower, "утечк|распространен.*запис|публикаци.*видео|доступ.*посторонн|незаконн.*доступ.*запис")
  pat_operational_violation <- str_detect(text_lower, "несвоевременн.*обслуживан|не проводится.*обслуживан|нарушени.*эксплуатаци|отсутствие.*техобслуживан")
  n_violations <- sum(pat_malfunction, pat_privacy, pat_storage, pat_evidence, 
                      pat_no_act, pat_absence_camera, pat_improper_placement, 
                      pat_no_consent, pat_violation_sanpin, pat_failure_to_install, 
                      pat_data_leak, pat_operational_violation, na.rm = TRUE)
  
  # 9. Количество упоминаний
  n_camera   <- str_count(text_lower, "видеонаблюдени")
  n_kamera   <- str_count(text_lower, "камер")
  n_school   <- str_count(text_lower, "школ")
  n_children <- str_count(text_lower, "несовершеннолетн|школьник|учащ")
  
  tibble(
    id = cid, 
    district = district, 
    is_valid = TRUE,
    court = court, 
    case_type = case_type,
    date_in = date_in, 
    date_out = date_out,
    text_length = text_length,
    kw_camera = kw_camera, kw_school = kw_school, kw_children = kw_children,
    kw_incident = kw_incident, kw_absence = kw_absence, kw_obligate = kw_obligate,
    n_camera = n_camera, n_kamera = n_kamera, n_school = n_school, n_children = n_children,
    pat_malfunction = pat_malfunction,
    pat_privacy = pat_privacy,
    pat_storage = pat_storage,
    pat_evidence = pat_evidence,
    pat_no_act = pat_no_act,
    pat_absence_camera = pat_absence_camera,
    pat_improper_placement = pat_improper_placement,
    pat_no_consent = pat_no_consent,
    pat_violation_sanpin = pat_violation_sanpin,
    pat_failure_to_install = pat_failure_to_install,
    pat_data_leak = pat_data_leak,
    pat_operational_violation = pat_operational_violation,
    n_violations = n_violations
  )
}

# Безопасное выполнение JS в браузере (с защитой от ошибок)
eval_js <- function(session, js) {
  res <- tryCatch(
    session$Runtime$evaluate(js, returnByValue = TRUE),
    error = function(e) list(result = NULL)
  )
  if (!is.null(res$result) && !is.null(res$result$value)) {
    return(res$result$value)
  }
  return(NULL)
}

# Навигация с повторами и увеличенным таймаутом
safe_get <- function(session, url, retries = MAX_RETRIES) {
  for (attempt in seq_len(retries)) {
    ok <- tryCatch({
      # Увеличиваем таймаут до 60 секунд (60000 мс)
      session$Page$navigate(url, timeout = 60000)
      Sys.sleep(4) # Даем больше времени на первоначальную отрисовку DOM
      TRUE
    }, error = function(e) {
      cat(sprintf("    Попытка %d не удалась: %s\n", attempt, conditionMessage(e)))
      FALSE
    })
    if (ok) return(TRUE)
    if (attempt < retries) Sys.sleep(RETRY_DELAY)
  }
  FALSE
}

get_html <- function(session, retries = 5) {
  for (i in seq_len(retries)) {
    res <- tryCatch(
      session$Runtime$evaluate(
        "document.documentElement.outerHTML",
        returnByValue = TRUE
      ),
      error = function(e) NULL
    )
    
    # ЖЕСТКАЯ ПРОВЕРКА: гарантируем, что вернется строка, а не NULL
    if (!is.null(res) && 
        !is.null(res$result) && 
        !is.null(res$result$value) && 
        is.character(res$result$value) && 
        nzchar(res$result$value)) {
      return(res$result$value)
    }
    Sys.sleep(1)
  }
  stop("Не удалось получить HTML страницы после нескольких попыток (вернулся NULL или пустая строка)")
}

  cat("Ожидание загрузки дерева судов (до 60 секунд)...\n")
  tree_loaded <- FALSE
  for (i in 1:60) {
    # Проверяем наличие элементов дерева
    n_lines <- eval_js(session, "document.querySelectorAll('div.aciTreeLine').length")

    tree_has_label <- eval_js(session, sprintf("
      (function() {
        let lines = document.querySelectorAll('div.aciTreeLine');
        for (let line of lines) {
          let txt = line.querySelector('.aciTreeText');
          if (txt && txt.textContent.trim() === %s) {
            return true;
          }
        }
        return false;
      })()
    ", label_json))
    
    if (!is.null(n_lines) && n_lines > 0 && !is.null(tree_has_label) && tree_has_label == TRUE) {
      cat(sprintf("Дерево загружено (найдено %d элементов, округ найден). Попытка %d.\n", n_lines, i))
      tree_loaded <- TRUE
      break
    }
    
    if (i %% 10 == 0) {
      cat(sprintf("  ... ожидание %d секунд (найдено %d элементов дерева)\n", i, ifelse(is.null(n_lines), 0, n_lines)))
    }
    Sys.sleep(1)

  if (!tree_loaded) {
    # Сохраняем HTML для диагностики
    debug_html <- get_html(session)
    writeLines(debug_html, file.path(BASE_DIR, "debug_tree_not_loaded.html"))

                 "3. Структура дерева изменилась"))
  }
  
  # БЕЗОПАСНАЯ передача строки в JS через jsonlite
  label_json <- jsonlite::toJSON(label_text, auto_unbox = TRUE)
  
  js <- sprintf("
    (function() {
      let lines = document.querySelectorAll('div.aciTreeLine');
      for (let line of lines) {
        let txt = line.querySelector('.aciTreeText');
        if (txt && txt.textContent.trim() === %s) {
          let push = line.querySelector('.aciTreePush');
          if (push) { push.click(); }
          let check = line.querySelector('.aciTreeCheck');
          if (check && !check.classList.contains('aciTreeChecked')) {
            check.click();
          }
          return 'ok';
        }
      }
      return 'not_found';
    })()
  ", label_json)
  
  res <- eval_js(session, js)
  if (is.null(res) || res != "ok") {
    stop(sprintf("Не найден узел дерева с текстом '%s'.", label_text))
  }
  Sys.sleep(2)
  
  # Дополнительно кликаем все дочерние чекбоксы, которые могли появиться
  js2 <- "
    (function() {
      let checks = document.querySelectorAll('.aciTreeCheck');
      let n = 0;
      for (let c of checks) {
        if (!c.classList.contains('aciTreeChecked')) {
          c.click();
          n++;
        }
      }
      return n;
    })()
  "
  eval_js(session, js2)
  Sys.sleep(1)
  cat(sprintf("Выбраны все суды/регионы для: %s\n", label_text))
}

# Конвертация одного HTML в PDF
convert_single_case <- function(session, html_path, delete_html = FALSE) {
  pdf_path <- sub("\\.html$", ".pdf", html_path)
  abs_path <- normalizePath(html_path, mustWork = FALSE)
  file_url <- paste0("file:///", gsub("\\\\", "/", abs_path))
  tryCatch({
    session$Page$navigate(file_url)
    Sys.sleep(1)
    res <- session$Page$printToPDF(
      printBackground = TRUE,
      paperWidth      = 8.27,
      paperHeight     = 11.69,
      marginTop       = 0.4,
      marginBottom    = 0.4,
      marginLeft      = 0.4,
      marginRight     = 0.4
    )
    writeBin(base64decode(res$data), pdf_path)
    if (delete_html) file.remove(html_path)
    TRUE
  }, error = function(e) {
    cat(sprintf("    Ошибка конвертации %s: %s\n", html_path, conditionMessage(e)))
    FALSE
  })
}

# Ожидание реальной загрузки контента
wait_for_real_content <- function(session, max_seconds = 50) { # было 10 или 8
  for (i in 1:max_seconds) {
    res <- tryCatch(
      session$Runtime$evaluate("document.documentElement.outerHTML.length", returnByValue = TRUE),
      error = function(e) NULL
    )
    if (!is.null(res) && !is.null(res$result$value) && res$result$value > 5000) {
      return(TRUE)
    }
    Sys.sleep(1)
  }
  return(FALSE)
}

# Безопасная навигация
safe_get <- function(session, url, retries = 3) {
  for (attempt in seq_len(retries)) {
    ok <- tryCatch({
      session$Page$navigate(url)
      Sys.sleep(4) # Больше времени на отклик
      TRUE
    }, error = function(e) FALSE)
    if (ok) return(TRUE)
    if (attempt < retries) Sys.sleep(3)
  }
  FALSE
}
is_bad_file <- function(file_path) {
  f_size <- file.info(file_path)$size
  if (is.na(f_size) || f_size < MIN_FILE_SIZE) return(TRUE)
  
  # Читаем ТОЛЬКО первые 20 строк для быстрой проверки (оптимизация для больших файлов)
  lines <- tryCatch(
    readLines(file_path, warn = FALSE, encoding = "UTF-8", n = 20),
    error = function(e) character(0)
  )
  if (length(lines) == 0) return(TRUE)
  
  header_text <- tolower(paste(lines, collapse = " "))
  
  # 1. ПРОВЕРКА НА СТРАНИЦУ ОШИБКИ БРАУЗЕРА (Chromium/Chrome/Edge)
  # Эти страницы могут весить сотни КБ, но они являются мусором
  browser_error_patterns <- c(
    "err_", "chromium", "no internet", "name not resolved", 
    "connection refused", "connection reset", "net::err"
  )
  for (patt in browser_error_patterns) {
    if (str_detect(header_text, patt)) return(TRUE)
  }
  
  # 2. ПРОВЕРКА НА КАПЧУ / БЛОКИРОВКУ / АНТИБОТ
  block_patterns <- c(
    "cloudflare", "captcha", "проверка браузера", "доступ ограничен",
    "access denied", "just a moment", "ddos protection"
  )
  for (patt in block_patterns) {
    if (str_detect(header_text, patt)) return(TRUE)
  }
  
  # 3. ПРОВЕРКА НА ПУСТУЮ ОБОЛОЧКУ САЙТА
  # Если в первых 20 строках нет тега <body>, значит HTML битый или незагруженный
  if (!str_detect(header_text, "<body")) return(TRUE)
  
  # 4. ПРОВЕРКА НА СЛИШКОМ МАЛЕНЬКИЙ ФАЙЛ
  # Реальное решение суда с таблицами и текстом всегда > 25000 символов
  if (f_size < 25000) return(TRUE)
  
  return(FALSE)
}

  cat(sprintf("Найдено %d файлов со списками ID.\n", length(json_files)))
  
  # ОПТИМИЗАЦИЯ 3: Используем списки для накопления результатов, 
  # чтобы избежать квадратичной деградации скорости при использовании c()
  ids_to_redownload <- list()
  districts_for_ids <- list()
  
  for (json_file in json_files) {
    district <- str_extract(basename(json_file), "(?<=case_ids_).*?(?=\\.json)")
    target_folder <- file.path(BASE_DIR, paste0(district, "_Videonablyudenie_v_skole"))
    
    cat(sprintf("\n[Проверка] Округ: %s\n", district))
    
    all_ids <- fromJSON(json_file)
    if (is.list(all_ids)) all_ids <- unlist(all_ids, use.names = FALSE)
    
    n_ids <- length(all_ids)
    cat(sprintf("  Всего ID в файле: %d. Проверка...\n", n_ids))
    
    # Векторизованная проверка путей
    file_paths <- file.path(target_folder, paste0(all_ids, ".html"))
    exists_mask <- file.exists(file_paths)
    is_bad <- logical(n_ids)
    
    for (j in seq_len(n_ids)) {
      if (!exists_mask[j]) {
        is_bad[j] <- TRUE
      } else {
        is_bad[j] <- is_bad_file(file_paths[j])
        if (is_bad[j]) {
          file.remove(file_paths[j]) # Удаляем битый файл сразу
        }
      }
      
      # Индикатор прогресса каждые 500 файлов
      if (j %% 500 == 0) {
        cat(sprintf("    Проверено: %d / %d\n", j, n_ids))
      }
    }
    
    missing_count <- sum(!exists_mask)
    too_small_count <- sum(is_bad & exists_mask)
    
    cat(sprintf("  ИТОГО: Проверено: %d | Отсутствует: %d | Битых (удалено): %d\n", 
                n_ids, missing_count, too_small_count))
    
    # Безопасное добавление в список без деградации производительности
    if (any(is_bad)) {
      ids_to_redownload <- c(ids_to_redownload, list(all_ids[is_bad]))
      districts_for_ids <- c(districts_for_ids, list(rep(district, sum(is_bad))))
    }
  }
  
  # Распаковываем списки в плоские векторы только один раз в конце
  ids_to_redownload <- unlist(ids_to_redownload, use.names = FALSE)
  districts_for_ids <- unlist(districts_for_ids, use.names = FALSE)
  
  total_to_fix <- length(ids_to_redownload)
  cat(sprintf("\n=== ИТОГ ПРОВЕРКИ: Найдено %d файлов для перезагрузки ===\n", total_to_fix))
  
  if (total_to_fix == 0) {
    cat("Отлично! Все файлы загружены корректно и имеют достаточный размер.\n")
    return(invisible())
  }
  
  # 3. Процесс перезагрузки
  cat("\nЗапуск браузера для дозагрузки...\n")
  session <- ChromoteSession$new()
  on.exit(session$close(), add = TRUE)
  
  success_count <- 0
  failed_count <- 0
  
  # Предвыделяем память для лога ошибок
  failed_log <- character(total_to_fix)
  
  for (i in seq_len(total_to_fix)) {
    cid <- ids_to_redownload[i]
    district <- districts_for_ids[i]
    url <- paste0(BASE_URL, "/", cid)
    target_folder <- file.path(BASE_DIR, paste0(district, "_Videonablyudenie_v_skole"))
    file_path <- file.path(target_folder, paste0(cid, ".html"))
    
    dir.create(target_folder, showWarnings = FALSE, recursive = TRUE)
    
    cat(sprintf("[%d/%d] ID: %s (%s) ... ", i, total_to_fix, cid, district))
    
    if (safe_get(session, url)) {
      # явно указываем 20 секунд ожидания
      content_loaded <- wait_for_real_content(session, max_seconds = 50) 
      
      if (content_loaded) {
        res <- tryCatch(
          session$Runtime$evaluate("document.documentElement.outerHTML", returnByValue = TRUE),
          error = function(e) list(result = list(value = NULL))
        )
        html_content <- res$result$value
        
        if (!is.null(html_content)) {
          # Используем тот же оптимизированный подход с одним regex
          has_error <- grepl(ERROR_REGEX, html_content, ignore.case = TRUE)
          
          if (has_error) {
            cat("FAIL (Ошибка в контенте)\n")
            failed_count <- failed_count + 1
            failed_log[failed_count] <- cid
          } else if (nchar(html_content) >= MIN_FILE_SIZE) {
            writeLines(html_content, file_path, useBytes = TRUE)
            cat("OK (", nchar(html_content), "симв.)\n")
            success_count <- success_count + 1
          } else {
            cat("FAIL (Слишком короткий)\n")
            failed_count <- failed_count + 1
            failed_log[failed_count] <- cid
          }
        } else {
          cat("FAIL (Пустой ответ)\n")
          failed_count <- failed_count + 1
          failed_log[failed_count] <- cid
        }
      } else {
        cat("FAIL (Таймаут)\n")
        failed_count <- failed_count + 1
        failed_log[failed_count] <- cid
      }
    } else {
      cat("FAIL (Ошибка навигации)\n")
      failed_count <- failed_count + 1
      failed_log[failed_count] <- cid
    }
    
    Sys.sleep(1.5) # Пауза между запросами
  }
  
  cat("\n=== ОТЧЕТ О ВОССТАНОВЛЕНИИ ===\n")
  cat(sprintf("Успешно перезагружено: %d файлов\n", success_count))
  cat(sprintf("Не удалось загрузить: %d файлов\n", failed_count))
  
  if (failed_count > 0) {
    log_file <- file.path(BASE_DIR, "permanently_failed_ids.txt")
    writeLines(failed_log[1:failed_count], log_file)
    cat(sprintf("Список окончательно неудачных ID сохранен в: %s\n", log_file))
  }
  
  cat("Процесс завершен.\n")
}

main_repair()
